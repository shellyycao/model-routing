```python
import os
import re
import json
import math
from datetime import datetime

import pandas as pd

MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

REGION_MAP = {
    "north": "North", "south": "South", "east": "East", "west": "West",
    "n": "North", "s": "South", "e": "East", "w": "West",
}

CATEGORY_MAP = {
    "electronics": "Electronics", "electronic": "Electronics",
    "clothing": "Clothing", "apparel": "Clothing",
    "home & garden": "Home & Garden", "home and garden": "Home & Garden",
    "home&garden": "Home & Garden",
    "sports": "Sports", "sport": "Sports",
    "books": "Books", "book": "Books",
}

DATE_PATTERNS = [
    (re.compile(r"^\d{4}-\d{2}-\d{2}$"), "%Y-%m-%d"),
    (re.compile(r"^\d{2}/\d{2}/\d{4}$"), "%m/%d/%Y"),
    (re.compile(r"^\d{2}-[A-Za-z]{3}-\d{4}$"), "%d-%b-%Y"),
    (re.compile(r"^\d{4}/\d{2}/\d{2}$"), "%Y/%m/%d"),
    (re.compile(r"^[A-Za-z]+ \d{1,2}, \d{4}$"), "%B %d, %Y"),
]


def clean_text(value):
    if value is None:
        return None
    if isinstance(value, float) and math.isnan(value):
        return None
    s = str(value).strip()
    if s.lower() in MISSING_TOKENS:
        return None
    return s


def parse_order_date(s):
    for pattern, fmt in DATE_PATTERNS:
        if pattern.match(s):
            try:
                return datetime.strptime(s, fmt).strftime("%Y-%m-%d")
            except ValueError:
                return None
    return None


def parse_unit_price(s):
    t = s.replace("$", "").replace("USD", "").replace(",", "").strip()
    if not t:
        return None
    try:
        v = float(t)
    except ValueError:
        return None
    if not math.isfinite(v) or v <= 0 or v > 5000:
        return None
    return v


def parse_quantity(s):
    try:
        v = float(s)
    except ValueError:
        return None
    if not math.isfinite(v) or not v.is_integer():
        return None
    q = int(v)
    if q < 1 or q > 50:
        return None
    return q


def parse_discount(s):
    if s is None:
        return 0.0
    t = s
    if t.endswith("%"):
        t = t[:-1].strip()
    try:
        v = float(t)
    except ValueError:
        return None
    if not math.isfinite(v) or v < 0 or v > 100:
        return None
    return v


def main():
    input_dir = os.environ["INPUT_DIR"]
    output_dir = os.environ["OUTPUT_DIR"]
    os.makedirs(output_dir, exist_ok=True)

    raw_path = os.path.join(input_dir, "raw.csv")
    df = pd.read_csv(raw_path, dtype=str, keep_default_na=False, na_filter=False)

    summary = {"raw_rows": int(len(df))}
    counts = {
        "duplicate_order_id_removed": 0,
        "dropped_missing_order_id": 0,
        "dropped_invalid_order_date": 0,
        "dropped_invalid_region": 0,
        "dropped_invalid_product_category": 0,
        "dropped_invalid_unit_price": 0,
        "dropped_invalid_quantity": 0,
        "dropped_invalid_discount_pct": 0,
        "discount_pct_missing_set_to_0": 0,
        "customer_id_missing_set_to_UNKNOWN": 0,
    }

    # R1: strip + normalize missing tokens
    records = df.to_dict("records")
    cleaned_rows = []
    seen_order_ids = set()

    for rec in records:
        row = {k: clean_text(v) for k, v in rec.items()}

        order_id = row.get("order_id")
        if order_id is None:
            counts["dropped_missing_order_id"] += 1
            continue
        # R2: dedupe before any other row filter
        if order_id in seen_order_ids:
            counts["duplicate_order_id_removed"] += 1
            continue
        seen_order_ids.add(order_id)

        # R3
        od = row.get("order_date")
        if od is None:
            counts["dropped_invalid_order_date"] += 1
            continue
        od_parsed = parse_order_date(od)
        if od_parsed is None:
            counts["dropped_invalid_order_date"] += 1
            continue

        # R4
        rg = row.get("region")
        region = REGION_MAP.get(rg.lower()) if rg is not None else None
        if region is None:
            counts["dropped_invalid_region"] += 1
            continue

        # R5
        pc = row.get("product_category")
        category = CATEGORY_MAP.get(pc.lower()) if pc is not None else None
        if category is None:
            counts["dropped_invalid_product_category"] += 1
            continue

        # R6
        up = row.get("unit_price")
        if up is None:
            counts["dropped_invalid_unit_price"] += 1
            continue
        price = parse_unit_price(up)
        if price is None:
            counts["dropped_invalid_unit_price"] += 1
            continue

        # R7
        qt = row.get("quantity")
        if qt is None:
            counts["dropped_invalid_quantity"] += 1
            continue
        qty = parse_quantity(qt)
        if qty is None:
            counts["dropped_invalid_quantity"] += 1
            continue

        # R8
        dc = row.get("discount_pct")
        if dc is None:
            counts["discount_pct_missing_set_to_0"] += 1
        disc = parse_discount(dc)
        if disc is None:
            counts["dropped_invalid_discount_pct"] += 1
            continue

        # R9
        cid = row.get("customer_id")
        if cid is None:
            counts["customer_id_missing_set_to_UNKNOWN"] += 1
            customer_id = "UNKNOWN"
        else:
            customer_id = cid.upper()

        # R10
        revenue = round(qty * price * (1 - disc / 100), 2)

        cleaned_rows.append({
            "order_id": order_id,
            "order_date": od_parsed,
            "customer_id": customer_id,
            "region": region,
            "product_category": category,
            "quantity": qty,
            "unit_price": price,
            "discount_pct": disc,
            "revenue": revenue,
        })

    out_df = pd.DataFrame(cleaned_rows, columns=[
        "order_id", "order_date", "customer_id", "region",
        "product_category", "quantity", "unit_price", "discount_pct", "revenue",
    ])
    out_df = out_df.sort_values("order_id", ascending=True).reset_index(drop=True)

    out_path = os.path.join(output_dir, "cleaned.csv")

    summary.update(counts)
    summary["cleaned_rows"] = int(len(out_df))

    summary_path = os.path.join(output_dir, "cleaning_summary.json")
    with open(summary_path, "w") as f:
        json.dump(summary, f, indent=2)


if __name__ == "__main__":
    main()
```

```python
def clean_text(value):
    """R1: strip whitespace; map missing tokens (case-insensitive) to None."""
    if value is None:
        return None
    if isinstance(value, float) and math.isnan(value):
        return None
    s = str(value).strip()
    if s.lower() in MISSING_TOKENS:
        return None
    return s


def parse_order_date(s):
    """R3: accept exactly the five allowed date formats; return YYYY-MM-DD."""
    for pattern, fmt in DATE_PATTERNS:
        if pattern.match(s):
            try:
                return datetime.strptime(s, fmt).strftime("%Y-%m-%d")
            except ValueError:
                return None
    return None


def parse_unit_price(s):
    """R6: strip $ / USD / thousands separators; require 0 < price <= 5000."""
    t = s.replace("$", "").replace("USD", "").replace(",", "").strip()
    if not t:
        return None
    try:
        v = float(t)
    except ValueError:
        return None
    if not math.isfinite(v) or v <= 0 or v > 5000:
        return None
    return v


def parse_quantity(s):
    """R7: numeric whole-number quantity in [1, 50]."""
    try:
        v = float(s)
    except ValueError:
        return None
    if not math.isfinite(v) or not v.is_integer():
        return None
    q = int(v)
    if q < 1 or q > 50:
        return None
    return q


def parse_discount(s):

        return None
    return v
```
