```python
import json
import os

import numpy as np
import pandas as pd


def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")
    output_path = os.path.join(os.environ["OUTPUT_DIR"], "findings.json")

    df = pd.read_csv(input_path)

    # Ensure expected dtypes
    df["signup_date"] = pd.to_datetime(df["signup_date"], format="%Y-%m-%d")
    # satisfaction_score may be missing -> float with NaN

    findings = {}

    def rate_pct(count, total):
        if total == 0:
            return 0.0
        return float(count) / float(total) * 100.0

    # q1
    customer_count = int(len(df))
    findings["customer_count"] = customer_count
    findings["churn_rate_pct"] = rate_pct((df["churned"] == 1).sum(), customer_count)
    findings["avg_monthly_fee"] = float(df["monthly_fee"].mean())

    # q2 churn by plan
    for plan, key in [("Basic", "churn_rate_basic_pct"),
                      ("Standard", "churn_rate_standard_pct"),
                      ("Premium", "churn_rate_premium_pct")]:
        sub = df[df["plan"] == plan]
        findings[key] = rate_pct((sub["churned"] == 1).sum(), len(sub))

    # q3 cohorts by signup year
    years = df["signup_date"].dt.year
    for year in (2021, 2022, 2023, 2024):
        sub = df[years == year]
        findings[f"churn_rate_cohort_{year}_pct"] = rate_pct((sub["churned"] == 1).sum(), len(sub))

    # q4 contract effect
    monthly = df[df["contract_type"] == "Monthly"]
    annual = df[df["contract_type"] == "Annual"]
    churn_monthly = rate_pct((monthly["churned"] == 1).sum(), len(monthly))
    churn_annual = rate_pct((annual["churned"] == 1).sum(), len(annual))
    findings["churn_rate_monthly_pct"] = churn_monthly
    findings["churn_rate_annual_pct"] = churn_annual
    if churn_annual == 0.0:
        findings["churn_monthly_vs_annual_change_pct"] = 0.0
    else:
        findings["churn_monthly_vs_annual_change_pct"] = (churn_monthly - churn_annual) / churn_annual * 100.0

    # q5 at-risk segment
    sat = df["satisfaction_score"]
    mask = (df["support_tickets"] >= 5) & sat.notna() & (sat <= 2)
    at_risk = df[mask]
    findings["at_risk_customer_count"] = int(len(at_risk))
    findings["at_risk_churn_rate_pct"] = rate_pct((at_risk["churned"] == 1).sum(), len(at_risk))

    # q6 tenure and satisfaction summary
    tenure = df["tenure_months"]
    findings["tenure_mean"] = float(tenure.mean())
    findings["tenure_median"] = float(tenure.median())
    findings["tenure_std"] = float(tenure.std(ddof=1))
    findings["satisfaction_mean"] = float(sat.dropna().mean())
    findings["satisfaction_missing_count"] = int(sat.isna().sum())

    # q7 LTV by plan
    ltv = {}
    for plan, key in [("Basic", "ltv_basic"), ("Standard", "ltv_standard"), ("Premium", "ltv_premium")]:
        sub = df[df["plan"] == plan]
        val = float(sub["lifetime_value"].mean()) if len(sub) else 0.0
        ltv[plan] = val
        findings[key] = val
    if ltv["Basic"] == 0.0:
        findings["premium_vs_basic_ltv_change_pct"] = 0.0
    else:
        findings["premium_vs_basic_ltv_change_pct"] = (ltv["Premium"] - ltv["Basic"]) / ltv["Basic"] * 100.0

    # Guarantee all values are finite numbers
    clean = {}
    for k, v in findings.items():
        fv = float(v)
        if not np.isfinite(fv):
            fv = 0.0
        clean[k] = fv

    with open(output_path, "w") as f:
        json.dump(clean, f, indent=2)


if __name__ == "__main__":
    main()
```
