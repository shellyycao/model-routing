def parse_signup_date(s):
    if s is None:
        return None
    s = s.strip()
    # YYYY-MM-DD
    m = re.fullmatch(r'(\d{4})-(\d{1,2})-(\d{1,2})', s)
    if m:
        y, mo, d = map(int, m.groups())
        return safe_date(y, mo, d)
    # MM/DD/YYYY
    m = re.fullmatch(r'(\d{1,2})/(\d{1,2})/(\d{4})', s)
    if m:
        mo, d, y = map(int, m.groups())
        return safe_date(y, mo, d)
    # DD-Mon-YYYY
    m = re.fullmatch(r'(\d{1,2})-([A-Za-z]{3})-(\d{4})', s)
    if m:
        d = int(m.group(1)); mon = MONTHS.get(m.group(2).lower()); y = int(m.group(3))
        if mon:
            return safe_date(y, mon, d)
        return None
    # YYYY/MM/DD
    m = re.fullmatch(r'(\d{4})/(\d{1,2})/(\d{1,2})', s)
    if m:
        y, mo, d = map(int, m.groups())
        return safe_date(y, mo, d)
    # Month D, YYYY
    m = re.fullmatch(r"([A-Za-z]+)\s+(\d{1,2}),\s*(\d{4})", s)
    if m:
        mon = MONTHS.get(m.group(1).lower()); d = int(m.group(2)); y = int(m.group(3))
        if mon:
