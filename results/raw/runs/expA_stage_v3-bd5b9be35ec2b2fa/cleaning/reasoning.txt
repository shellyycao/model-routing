import os, pandas as pd, numpy as np, json, re, datetime

input_path = os.path.join(os.environ['INPUT_DIR'], 'raw.csv')
df = pd.read_csv(input_path, dtype=str, keep_default_na=False) # keep as strings

# strip whitespace
df = df.applymap(lambda x: x.strip() if isinstance(x, str) else x)

# treat missing values
missing_vals = {'', 'na', 'n/a', 'null', 'none', '-'}
df = df.replace(to_replace=missing_vals, value=np.nan, regex=False)

raw_rows = len(df)

# R2 dedup
df = df.drop_duplicates(subset='order_id', keep='first')
dedup_rows = len(df)

# Prepare list
cleaned = []
counts = {
    'raw_rows': raw_rows,
    'after_dedup_rows': dedup_rows,
    'date_dropped':0,
    'region_dropped':0,
    'category_dropped':0,
    'price_dropped':0,
    'quantity_dropped':0,
    'discount_dropped':0,
    'cleaned_rows':0
}

for idx, row in df.iterrows():
    # order_id must not be missing
    if pd.isna(row['order_id']):
        continue  # drop (unlikely)
    # R3 order_date
    od = row['order_date']
    if pd.isna(od):
        counts['date_dropped']+=1
        continue
    parsed_date = None
    for fmt in ["%Y-%m-%d","%m/%d/%Y","%d-%b-%Y","%Y/%m/%d","%B %d, %Y"]:
        try:
            # For %d-%b-%Y ensure month abbreviation capitalized
            if fmt == "%d-%b-%Y":
                od_try = od.title()
            else:
                od_try = od
            parsed_date = datetime.datetime.strptime(od_try, fmt).date()
            break
        except Exception:
            continue
    if not parsed_date:
        counts['date_dropped']+=1
        continue
    order_date_str = parsed_date.isoformat()
    # R4 region
    region_raw = row['region']
    if pd.isna(region_raw):
        counts['region_dropped']+=1
        continue
    region_map = {
        'n':'North','north':'North',
        's':'South','south':'South',
        'e':'East','east':'East',
        'w':'West','west':'West'
    }
    region_key = region_raw.strip().lower()
    region = region_map.get(region_key)
    if not region:
        counts['region_dropped']+=1
        continue
    # R5 product_category
    cat_raw = row['product_category']
    if pd.isna(cat_raw):
        counts['category_dropped']+=1
        continue
    cat_key = cat_raw.strip().lower()
    cat_map = {
        'electronics':'Electronics',
        'electronic':'Electronics',
        'clothing':'Clothing',
        'apparel':'Clothing',
        'home & garden':'Home & Garden',
        'home and garden':'Home & Garden',
        'home&garden':'Home & Garden',
        'sports':'Sports',
        'sport':'Sports',
        'books':'Books',
        'book':'Books'
    }
    product_category = cat_map.get(cat_key)
    if not product_category:
        counts['category_dropped']+=1
        continue
    # R6 unit_price
    up_raw = row['unit_price']
    if pd.isna(up_raw):
        counts['price_dropped']+=1
        continue
    up_clean = re.sub(r'[\$,]', '', up_raw, flags=re.IGNORECASE)
    up_clean = re.sub(r'(?i)usd', '', up_clean)
    up_clean = up_clean.strip()
    try:
        unit_price = float(up_clean)
    except:
        counts['price_dropped']+=1
        continue
    if unit_price <=0 or unit_price > 5000:
        counts['price_dropped']+=1
        continue
    # R7 quantity
    qty_raw = row['quantity']
    if pd.isna(qty_raw):
        counts['quantity_dropped']+=1
        continue
    try:
        qty_float = float(qty_raw)
    except:
        counts['quantity_dropped']+=1
        continue
    if not qty_float.is_integer():
        counts['quantity_dropped']+=1
        continue
    quantity = int(qty_float)
    if quantity <1 or quantity >50:
        counts['quantity_dropped']+=1
        continue
    # R8 discount_pct
    disc_raw = row['discount_pct']
    if pd.isna(disc_raw):
        discount_pct = 0.0
    else:
        disc_clean = disc_raw.strip()
        if disc_clean.endswith('%'):
            disc_clean = disc_clean[:-1]
        try:
            discount_pct = float(disc_clean)
        except:
            counts['discount_dropped']+=1
            continue
    if discount_pct <0 or discount_pct >100:
        counts['discount_dropped']+=1
        continue
    # R9 customer_id
    cust_raw = row['customer_id']
    if pd.isna(cust_raw):
        customer_id = 'UNKNOWN'
    else:
        customer_id = str(cust_raw).upper()
    # R10 revenue
    revenue = round(quantity * unit_price * (1 - discount_pct/100),2)
    # Append cleaned row dict
    cleaned.append({
        'order_id': row['order_id'],
        'order_date': order_date_str,
        'customer_id': customer_id,
        'region': region,
        'product_category': product_category,
        'quantity': quantity,
        'unit_price': unit_price,
        'discount_pct': discount_pct,
        'revenue': revenue
    })
# after loop
cleaned_rows = len(cleaned)
counts['cleaned_rows'] = cleaned_rows

# create DataFrame
cleaned_df = pd.DataFrame(cleaned)
# sort by order_id ascending
cleaned_df = cleaned_df.sort_values('order_id')
# write CSV
output_path = os.path.join(os.environ['OUTPUT_DIR'], 'cleaned.csv')
cleaned_df.to_csv(output_path, index=False)

# write summary
summary_path = os.path.join(os.environ['OUTPUT_DIR'], 'cleaning_summary.json')
with open(summary_path, 'w') as f:
    json.dump(counts, f, indent=2)
