def parse_temperature(value):
    if pd.isna(value): return ("missing", None)
    text = str(value)
    match = TEMP_RE.match(text)
    if not match: return ("invalid", None)
    number_text, suffix = match.groups()
    compact_suffix = re.sub(r"\s+", "", suffix).lower()
    if compact_suffix in {"f", "°f"}: unit = "f"
    elif compact_suffix in {"", "c", "°c", "°"}: unit = "c"
    else: return ("invalid", None)
    try: raw_value = float(number_text)
    except ValueError: return ("invalid", None)
    if unit == "f": celsius = (raw_value - 32.0) * 5.0 / 9.0
    else: celsius = raw_value
    if not math.isfinite(celsius): return ("invalid", None)
    if celsius < -40.0 or celsius > 50.0: return ("out_of_range", None)
    return ("ok", round(celsius, 1))
