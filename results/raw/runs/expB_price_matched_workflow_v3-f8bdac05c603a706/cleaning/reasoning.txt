def parse_energy(value):
    """Return energy in kWh (rounded to 2 dp) or np.nan if invalid."""
    if pd.isna(value):
        return np.nan
    text = str(value).replace(",", "")
    multiplier = 1.0
    m = ENERGY_UNIT_RE.match(text)
    if m:
        number_text = m.group(1)
        if m.group(2).lower() == "mwh":
            multiplier = 1000.0
    else:
        number_text = text
    number_text = number_text.strip()
    if not number_text:
        return np.nan
    try:
        number = float(number_text)
    except ValueError:
        return np.nan
    if not math.isfinite(number):
        return np.nan
    number *= multiplier
    if number < 0 or number > 100000:
        return np.nan
    return round(number, 2)

def parse_temperature(value):

    if pd.isna(value):
