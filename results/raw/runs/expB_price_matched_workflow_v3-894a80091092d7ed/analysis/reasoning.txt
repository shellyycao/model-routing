```python
import json
import os

import numpy as np
import pandas as pd

def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")
    output_path = os.path.join(os.environ["OUTPUT_DIR"], "findings.json")

    df = pd.read_csv(input_path, dtype={"order_id": str, "customer_id": str})
    df["order_date"] = pd.to_datetime(df["order_date"], format="%Y-%m-%d")
    df["revenue"] = df["revenue"].astype(float)
    df["discount_pct"] = df["discount_pct"].astype(float)

    findings = {}

    # q1
    total_revenue = float(df["revenue"].sum())
    order_count = int(len(df))
    findings["total_revenue"] = total_revenue
    findings["order_count"] = order_count
    findings["average_order_value"] = total_revenue / order_count

    # q2
    by_region = df.groupby("region")["revenue"].sum()
    for key, region in [("revenue_north","North"),("revenue_south","South"),("revenue_east","East"),("revenue_west","West")]:
        findings[key] = float(by_region.get(region, 0.0))

    # q3
    df["year"] = df["order_date"].dt.year
    df["quarter"] = df["order_date"].dt.quarter
    d2024 = df[df["year"] == 2024]
    qsum = d2024.groupby("quarter")["revenue"].sum()
    for q in (1,2,3,4):
        findings[f"revenue_2024_q{q}"] = float(qsum.get(q, 0.0))

    # q4
    revenue_2023 = float(df.loc[df["year"] == 2023, "revenue"].sum())
    revenue_2024 = float(df.loc[df["year"] == 2024, "revenue"].sum())
    findings["revenue_2023"] = revenue_2023
    findings["revenue_2024"] = revenue_2024
    findings["revenue_yoy_change_pct"] = (revenue_2024 - revenue_2023) / revenue_2023 * 100.0 if revenue_2023 != 0 else 0.0

    # q5
    mask = (df["product_category"] == "Electronics") & (df["discount_pct"] >= 20)
    hd = df[mask]
    findings["electronics_high_discount_orders"] = int(len(hd))
    findings["electronics_high_discount_revenue"] = float(hd["revenue"].sum())

    # q6
    rev = df["revenue"]
    findings["order_revenue_mean"] = float(rev.mean())
    findings["order_revenue_median"] = float(rev.median())
    std = float(rev.std(ddof=1))
    findings["order_revenue_std"] = std if np.isfinite(std) else 0.0
    findings["order_revenue_min"] = float(rev.min())
    findings["order_revenue_max"] = float(rev.max())

    # q7
    by_cust = df.groupby("customer_id")["revenue"].sum()
    findings["unique_customer_count"] = int(df["customer_id"].nunique())
    top = float(by_cust.max()) if len(by_cust) else 0.0
    findings["top_customer_revenue"] = top
    findings["top_customer_revenue_share_pct"] = top / total_revenue * 100.0 if total_revenue != 0 else 0.0

    # sanity: ensure all finite
    for k, v in findings.items():
        if not isinstance(v, (int, float)) or not np.isfinite(v):
            raise ValueError(f"Non-finite metric {k}: {v}")

    os.makedirs(os.environ["OUTPUT_DIR"], exist_ok=True)  # maybe not needed; OUTPUT_DIR exists
    with open(output_path, "w") as f:
        json.dump(findings, f, allow_nan=False)

if __name__ == "__main__":
    main()
```
