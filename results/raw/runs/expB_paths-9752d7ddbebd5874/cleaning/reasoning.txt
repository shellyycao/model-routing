def parse_date(s):
    if pd.isna(s):
        return None
    for fmt in ["%Y-%m-%d","%m/%d/%Y","%d-%b-%Y","%Y/%m/%d","%B %d, %Y"]:
        try:
            dt = datetime.datetime.strptime(s, fmt)
            return dt.strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None
