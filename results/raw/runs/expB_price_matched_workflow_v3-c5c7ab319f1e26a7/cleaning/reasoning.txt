```python
import json
import os
import re
from datetime import datetime

import numpy as np
import pandas as pd

MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

DATE_FORMATS = (
    "%Y-%m-%d",   # YYYY-MM-DD
    "%m/%d/%Y",   # MM/DD/YYYY
    "%d-%b-%Y",   # DD-Mon-YYYY
    "%Y/%m/%d",   # YYYY/MM/DD
    "%B %d, %Y",  # Month D, YYYY
)

BUILDING_TYPE_MAP = {
    "residential": "Residential",
    "res": "Residential",
    "resi": "Residential",
    "commercial": "Commercial",
    "comm": "Commercial",
    "industrial": "Industrial",
    "ind": "Industrial",
}

TARIFF_MAP = {
    "peak": "peak",
    "off-peak": "off-peak",
    "offpeak": "off-peak",
    "off peak": "off-peak",
}

TEMP_RE = re.compile(r"^([+-]?(?:\d+(?:\.\d+)?|\.\d+))\s*°?\s*([CcFf])?$")


def parse_reading_date(value):
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(value, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None


def parse_energy(value):
    """Return energy in kWh (float) or None if unparseable."""
    s = value.replace(",", "").strip()
    factor = 1.0
    low = s.lower()
    if low.endswith("mwh"):
        factor = 1000.0
        s = s[:-3].strip()
    elif low.endswith("kwh"):
        s = s[:-3].strip()
    if not s:
        return None
    try:
        return float(s) * factor
    except ValueError:
        return None


def parse_temperature(value):
    """Return temperature in Celsius or None (row is kept either way)."""
    m = TEMP_RE.match(value.strip())
    if not m:
        return None
    try:
        t = float(m.group(1))
    except ValueError:
        return None
    if m.group(2) and m.group(2).lower() == "f":
        t = (t - 32.0) * 5.0 / 9.0
    if t < -40.0 or t > 50.0:
        return None
    return round(t, 1)


def main():
    input_dir = os.environ["INPUT_DIR"]
    output_dir = os.environ["OUTPUT_DIR"]
    raw_path = os.path.join(input_dir, "raw.csv")

    # R1: read everything as text, strip whitespace, normalise missing tokens.
    df = pd.read_csv(raw_path, dtype=str, keep_default_na=False, na_filter=False)
    raw_rows = len(df)
    df = df.apply(lambda col: col.astype(str).str.strip())
    df = df.apply(lambda col: col.map(lambda v: None if v.lower() in MISSING_TOKENS else v))

    summary = {"raw_rows": int(raw_rows)}

    # R2: drop rows with missing reading_id, then de-duplicate on reading_id (keep first).
    missing_id = int(df["reading_id"].isna().sum())
    df = df[df["reading_id"].notna()]
    before = len(df)
    df = df.drop_duplicates(subset="reading_id", keep="first")
    summary["r2_missing_reading_id_dropped"] = missing_id
    summary["r2_duplicate_reading_id_dropped"] = int(before - len(df))

    # R3: reading_date
    parsed = df["reading_date"].map(lambda v: parse_reading_date(v) if v is not None else None)
    bad = parsed.isna()
    summary["r3_reading_date_dropped"] = int(bad.sum())
    df = df.loc[~bad].copy()
    df["reading_date"] = parsed[~bad]

    # R4: meter_id
    bad = df["meter_id"].isna()
    summary["r4_meter_id_dropped"] = int(bad.sum())
    df = df.loc[~bad].copy()
    df["meter_id"] = df["meter_id"].str.upper()

    # R5: building_type
    bt = df["building_type"].map(lambda v: BUILDING_TYPE_MAP.get(v.lower()) if v is not None else None)
    bad = bt.isna()
    summary["r5_building_type_dropped"] = int(bad.sum())
    df = df.loc[~bad].copy()
    df["building_type"] = bt[~bad]

    # R6: tariff
    tf = df["tariff"].map(
        lambda v: TARIFF_MAP.get(re.sub(r"\s+", " ", v.lower())) if v is not None else None
    )
    bad = tf.isna()
    summary["r6_tariff_dropped"] = int(bad.sum())
    df = df.loc[~bad].copy()
    df["tariff"] = tf[~bad]

    # R7: energy -> energy_kwh
    en = df["energy"].map(lambda v: parse_energy(v) if v is not None else None)
    valid = en.notna() & en.map(lambda x: 0.0 <= x <= 100000.0 if x is not None else False)
    summary["r7_energy_dropped"] = int((~valid).sum())
    df = df.loc[valid].copy()
    df["energy_kwh"] = en[valid].map(lambda x: round(float(x), 2))

    # R8: temperature -> temperature_c (invalid -> empty, row kept)
    temp_raw = df["temperature"]
    tc = temp_raw.map(lambda v: parse_temperature(v) if v is not None else None)
    summary["r8_temperature_set_missing"] = int(tc.isna().sum())
    df["temperature_c"] = tc.astype("float64")  # None -> NaN

    # R9: final column order, sort, write
    out = df[["reading_id", "reading_date", "meter_id", "building_type",
              "tariff", "energy_kwh", "temperature_c"]].copy()
    out = out.sort_values("reading_id", kind="stable").reset_index(drop=True)
    out_path = os.path.join(output_dir, "cleaned.csv")
    out.to_csv(out_path, index=False)

    summary["cleaned_rows"] = int(len(out))
    with open(os.path.join(output_dir, "cleaning_summary.json"), "w") as fh:
        json.dump(summary, fh, indent=2)


if __name__ == "__main__":
    main()
```
