import json
import math
import os

import pandas as pd


def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")
    output_path = os.path.join(os.environ["OUTPUT_DIR"], "findings.json")

    df = pd.read_csv(input_path)

    findings = {}

    # q1
    ticket_count = int(len(df))
    findings["ticket_count"] = ticket_count
    findings["mean_resolution_minutes"] = float(df["resolution_minutes"].mean())
    findings["total_labor_cost"] = float(df["labor_cost"].sum())

    # q2
    team_cost = df.groupby("team", observed=True)["labor_cost"].sum()
    for team in ("Alpha", "Beta", "Gamma"):
        findings[f"labor_cost_{team.lower()}"] = float(team_cost.get(team, 0.0))

    # q3
    chan_res = df.groupby("channel", observed=True)["resolution_minutes"].mean()
    for ch in ("email", "chat", "phone"):
        findings[f"resolution_{ch}"] = float(chan_res.get(ch, 0.0))

    # q4
    reopen_by_channel = df.groupby("channel", observed=True)["reopened"].mean() * 100.0
    email_rate = float(reopen_by_channel.get("email", 0.0))
    chat_rate = float(reopen_by_channel.get("chat", 0.0))
    findings["reopen_rate_email_pct"] = email_rate
    findings["reopen_rate_chat_pct"] = chat_rate
    findings["email_vs_chat_reopen_change_pct"] = (
        (email_rate - chat_rate) / chat_rate * 100.0 if chat_rate != 0.0 else 0.0
    )

    # q5
    seg = df[(df["labor_cost"] >= 100) & (df["satisfaction_score"].notna()) & (df["satisfaction_score"] <= 2)]
    seg_count = int(len(seg))
    findings["high_cost_low_sat_count"] = seg_count
    findings["high_cost_low_sat_reopen_pct"] = float(seg["reopened"].mean() * 100.0) if seg_count else 0.0

    # q6
    findings["resolution_median"] = float(df["resolution_minutes"].median())
    findings["resolution_std"] = float(df["resolution_minutes"].std(ddof=1))
    findings["satisfaction_mean"] = float(df["satisfaction_score"].mean())
    findings["satisfaction_missing_count"] = int(df["satisfaction_score"].isna().sum())

    # q7
    agent_counts = df["agent_id"].value_counts()
    unique_agents = int(agent_counts.size) # or nunique
    top_count = int(agent_counts.iloc[0]) if unique_agents else 0
    findings["unique_agent_count"] = unique_agents
    findings["top_agent_ticket_count"] = top_count
    findings["top_agent_ticket_share_pct"] = top_count / ticket_count * 100.0 if ticket_count else 0.0

    # ensure finite
    for k, v in findings.items():
        fv = float(v)
        if not math.isfinite(fv):
            raise ValueError(...)
    # keep ints as ints? fine either way; write as is.

    os.makedirs(os.environ["OUTPUT_DIR"], exist_ok=True) — maybe not needed; output dir exists. Could still create. It says write nowhere else; making dir is fine.

    with open(output_path, "w") as f:
        json.dump(findings, f)

if __name__ == "__main__":
    main()
