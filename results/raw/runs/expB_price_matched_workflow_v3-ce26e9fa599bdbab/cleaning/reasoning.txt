import json
import math
import os
import re
from datetime import datetime

import pandas as pd

MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

DATE_FORMATS = (
    "%Y-%m-%d",   # YYYY-MM-DD
    "%m/%d/%Y",   # MM/DD/YYYY (month first)
    "%d-%b-%Y",   # DD-Mon-YYYY (e.g. 05-Mar-2024)
    "%Y/%m/%d",   # YYYY/MM/DD
    "%B %d, %Y",  # Month D, YYYY (e.g. March 5, 2024)
)

REGION_MAP = {
    "north": "North", "south": "South", "east": "East", "west": "West",
    "n": "North", "s": "South", "e": "East", "w": "West",
}

CATEGORY_MAP = {
    "electronics": "Electronics",
    "electronic": "Electronics",
    "clothing": "Clothing",
    "apparel": "Clothing",
    "home & garden": "Home & Garden",
    "home and garden": "Home & Garden",
    "home&garden": "Home & Garden",
    "sports": "Sports",
    "sport": "Sports",
    "books": "Books",
    "book": "Books",
}

OUTPUT_COLUMNS = [
    "order_id", "order_date", "customer_id", "region", "product_category",
    "quantity", "unit_price", "discount_pct", "revenue",
]


def is_missing(value):
    return value is None or str(value).strip().lower() in MISSING_TOKENS


def parse_date(value):
    if value is None:
        return None
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(value, fmt).strftime("%Y-%m-%d")
        except ValueError:
            pass
    return None


def parse_unit_price(value):
    if value is None:
        return None
    text = value.replace("$", "").replace(",", "")
    text = re.sub(r"usd", "", text, flags=re.IGNORECASE)
    text = re.sub(r"\s+", "", text)
    try:
        number = float(text)
    except ValueError:
        return None
    if not math.isfinite(number) or number <= 0 or number > 5000:
        return None
    return number


def parse_quantity(value):
    if value is None:
        return None
    try:
        number = float(value)
    except ValueError:
        return None
    if not math.isfinite(number) or not number.is_integer():
        return None
    if number < 1 or number > 50:
        return None
    return int(number)


def parse_discount(value):
    if value is None:
        return 0.0
    text = value.strip()
    if text.endswith("%"):
        text = text[:-1].strip()
    try:
        number = float(text)
    except ValueError:
        return None
    if not math.isfinite(number) or number < 0 or number > 100:
        return None
    return number


def main():
    input_dir = os.environ["INPUT_DIR"]
    output_dir = os.environ["OUTPUT_DIR"]
    os.makedirs(output_dir, exist_ok=True)

    raw_path = os.path.join(input_dir, "raw.csv")
    cleaned_path = os.path.join(output_dir, "cleaned.csv")
    summary_path = os.path.join(output_dir, "cleaning_summary.json")

    # R1: read all columns as text; strip whitespace; normalize missing tokens.
    df = pd.read_csv(raw_path, dtype=str, keep_default_na=False, na_filter=False)
    raw_rows = int(len(df))
    for col in df.columns:
        df[col] = df[col].map(lambda v: None if is_missing(v) else str(v).strip())

    # R2: drop duplicate order_id, keep first occurrence in raw-file order.
    dup_mask = df["order_id"].notna() & df.duplicated(subset=["order_id"], keep="first")
    duplicates_removed = int(dup_mask.sum())
    df = df.loc[~dup_mask].copy()

    # R3: order_date
    df["order_date"] = df["order_date"].map(parse_date)
    dropped_order_date = int(df["order_date"].isna().sum())
    df = df.loc[df["order_date"].notna()].copy()

    # R4: region
    df["region"] = df["region"].map(lambda v: REGION_MAP.get(v.lower()) if v is not None else None)
    dropped_region = int(df["region"].isna().sum())
    df = df.loc[df["region"].notna()].copy()

    # R5: product_category
    df["product_category"] = df["product_category"].map(lambda v: CATEGORY_MAP.get(v.lower()) if v is not None else None)
    dropped_product_category = int(df["product_category"].isna().sum())
    df = df.loc[df["product_category"].notna()].copy()

    # R6: unit_price
    df["unit_price"] = df["unit_price"].map(parse_unit_price)
    dropped_unit_price = int(df["unit_price"].isna().sum())
    df = df.loc[df["unit_price"].notna()].copy()

    # R7: quantity
    df["quantity"] = df["quantity"].map(parse_quantity)
    dropped_quantity = int(df["quantity"].isna().sum())
