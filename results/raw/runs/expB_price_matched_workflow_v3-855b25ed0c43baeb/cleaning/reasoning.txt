import json
import os
import re
from datetime import datetime

import numpy as np
import pandas as pd

INPUT_DIR = os.environ["INPUT_DIR"]
OUTPUT_DIR = os.environ["OUTPUT_DIR"]

MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

DATE_FORMATS = [
    "%Y-%m-%d",   # YYYY-MM-DD
    "%m/%d/%Y",   # MM/DD/YYYY (month first)
    "%d-%b-%Y",   # DD-Mon-YYYY
    "%Y/%m/%d",   # YYYY/MM/DD
    "%B %d, %Y",  # Month D, YYYY
]

BUILDING_TYPE_MAP = {
    "residential": "Residential",
    "res": "Residential",
    "resi": "Residential",
    "commercial": "Commercial",
    "comm": "Commercial",
    "industrial": "Industrial",
    "ind": "Industrial",
}

TARIFF_MAP = {
    "peak": "peak",
    "off-peak": "off-peak",
    "offpeak": "off-peak",
    "off peak": "off-peak",
}

ENERGY_RE = re.compile(
    r"^([+-]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][+-]?\d+)?)\s*(mwh|kwh)?$",
    re.IGNORECASE,
)

TEMP_RE = re.compile(
    r"^([+-]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][+-]?\d+)?)\s*°?\s*([fc])?$",
    re.IGNORECASE,
)


def parse_reading_date(value):
    if not isinstance(value, str):
        return None
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(value, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None


def parse_energy(value):
    """Return energy in kWh, or None if missing/invalid/out of range."""
    if not isinstance(value, str):
        return None
    text = value.replace(",", "").strip()
    match = ENERGY_RE.match(text)
    if match is None:
        return None
    number = float(match.group(1))
    if (match.group(2) or "").lower() == "mwh":
        number *= 1000.0
    if number < 0 or number > 100000:
        return None
    return round(number, 2)


def parse_temperature(value):
    """Return temperature in Celsius, or NaN if missing/invalid/out of range."""
    if not isinstance(value, str):
        return np.nan
    match = TEMP_RE.match(value)
    if match is None:
        return np.nan
    number = float(match.group(1))
    unit = match.group(2)
    if unit is not None and unit.lower() == "f":
        number = (number - 32.0) * 5.0 / 9.0
    if number < -40 or number > 50:
        return np.nan
    return round(number, 1)


def main():
    raw_path = os.path.join(INPUT_DIR, "raw.csv")
    df = pd.read_csv(raw_path, dtype=str, keep_default_na=False, na_filter=False,
                     encoding="utf-8-sig")

        df[col] = df[col].map(lambda v: np.nan if v.lower() in MISSING_TOKENS else v)
