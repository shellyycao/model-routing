```python
import json
import math
import os

import numpy as np
import pandas as pd


def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")
    output_path = os.path.join(os.environ["OUTPUT_DIR"], "findings.json")

    df = pd.read_csv(input_path)

    # Parse dates and numeric columns defensively (data is already cleaned,
    # but ensure correct dtypes for the computations).
    df["reading_date"] = pd.to_datetime(df["reading_date"], format="%Y-%m-%d", errors="coerce")
    df["energy_kwh"] = pd.to_numeric(df["energy_kwh"], errors="coerce")
    df["temperature_c"] = pd.to_numeric(df["temperature_c"], errors="coerce")

    def num(value):
        """Convert to a finite float; fall back to 0.0 if NaN/inf."""
        try:
            v = float(value)
        except (TypeError, ValueError):
            return 0.0
        if math.isnan(v) or math.isinf(v):
            return 0.0
        return v

    findings = {}

    # q1: overall
    total_kwh = num(df["energy_kwh"].sum())
    reading_count = int(len(df))
    findings["total_kwh"] = total_kwh
    findings["reading_count"] = reading_count
    findings["mean_kwh_per_reading"] = num(total_kwh / reading_count) if reading_count else 0.0

    # q2: by building type
    by_type = df.groupby("building_type")["energy_kwh"].sum()
    findings["kwh_residential"] = num(by_type.get("Residential", 0.0))
    findings["kwh_commercial"] = num(by_type.get("Commercial", 0.0))
    findings["kwh_industrial"] = num(by_type.get("Industrial", 0.0))

    # q3: quarters
    month = df["reading_date"].dt.month
    quarter = ((month - 1) // 3) + 1
    qsum = df.groupby(quarter)["energy_kwh"].sum()
    for i, key in enumerate(["kwh_q1", "kwh_q2", "kwh_q3", "kwh_q4"], start=1):
        findings[key] = num(qsum.get(i, 0.0))

    # q4: halves
    h1 = num(df.loc[month <= 6, "energy_kwh"].sum())
    h2 = num(df.loc[month >= 7, "energy_kwh"].sum())
    findings["kwh_h1"] = h1
    findings["kwh_h2"] = h2
    findings["kwh_h2_vs_h1_change_pct"] = num((h2 - h1) / h1 * 100.0) if h1 != 0.0 else 0.0

    # q5: hot commercial days
    hot_comm = df[(df["building_type"] == "Commercial") & (df["temperature_c"].notna()) & (df["temperature_c"] >= 25)]
    findings["hot_commercial_reading_count"] = int(len(hot_comm))
    findings["hot_commercial_mean_kwh"] = num(hot_comm["energy_kwh"].mean()) if len(hot_comm) else 0.0

    # q6: distribution
    findings["kwh_median"] = num(df["energy_kwh"].median())
    findings["kwh_std"] = num(df["energy_kwh"].std(ddof=1))
    temp_nonmiss = df["temperature_c"].dropna()
    findings["temperature_mean"] = num(temp_nonmiss.mean())
    findings["temperature_missing_count"] = int(df["temperature_c"].isna().sum())
    comm = df[(df["building_type"] == "Commercial") & (df["temperature_c"].notna())]
    if len(comm) >= 2 and comm["temperature_c"].nunique() > 1 and comm["energy_kwh"].nunique() > 1:
        corr = num(comm["temperature_c"].corr(comm["energy_kwh"], method="pearson"))
    else:
        corr = 0.0
    findings["commercial_temp_kwh_corr"] = corr

    # q7: tariff and meters
    by_tariff = df.groupby("tariff")["energy_kwh"].sum()
    kwh_peak = num(by_tariff.get("peak", 0.0))
    kwh_off_peak = num(by_tariff.get("off-peak", 0.0))
    findings["kwh_peak"] = kwh_peak
    findings["kwh_off_peak"] = kwh_off_peak
    findings["peak_kwh_share_pct"] = num(kwh_peak / total_kwh * 100.0) if total_kwh != 0.0 else 0.0
    findings["unique_meter_count"] = int(df["meter_id"].nunique())
    by_meter = df.groupby("meter_id")["energy_kwh"].sum()
    findings["highest_use_meter_kwh"] = num(by_meter.max()) if len(by_meter) else 0.0

    # Ensure flat dict of finite numbers only
    clean = {}
    for k, v in findings.items():
        if isinstance(v, bool):
            v = float(v)
        if isinstance(v, int):
            clean[k] = v
        else:
            clean[k] = num(v)

    os.makedirs(os.environ["OUTPUT_DIR"], exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(clean, f, indent=2)


if __name__ == "__main__":
    main()
```
