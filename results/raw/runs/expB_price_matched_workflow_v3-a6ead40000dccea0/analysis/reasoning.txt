import json
import os

import numpy as np
import pandas as pd


def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")

    df = pd.read_csv(input_path)

    # q1: Overall
    ticket_count = int(len(df))
    findings["ticket_count"] = ticket_count

    findings["total_labor_cost"] = float(df["labor_cost"].sum())

    # q2: Team comparison - total labor cost per team
    team_cost = df.groupby("team")["labor_cost"].sum()
    findings["labor_cost_alpha"] = float(team_cost.get("Alpha", 0.0))
    findings["labor_cost_beta"] = float(team_cost.get("Beta", 0.0))
    findings["labor_cost_gamma"] = float(team_cost.get("Gamma", 0.0))

    # q3: Channel comparison - mean resolution minutes by channel
    chan_res = df.groupby("channel")["resolution_minutes"].mean()
    findings["resolution_email"] = float(chan_res.get("email", 0.0))
    findings["resolution_chat"] = float(chan_res.get("chat", 0.0))
    findings["resolution_phone"] = float(chan_res.get("phone", 0.0))

    # q4: Reopening change
    email_mask = df["channel"] == "email"
    chat_mask = df["channel"] == "chat"
    reopen_rate_email = float(df.loc[email_mask, "reopened"].mean()) * 100.0
    reopen_rate_chat = float(df.loc[chat_mask, "reopened"].mean()) * 100.0
    findings["reopen_rate_email_pct"] = reopen_rate_email
    findings["reopen_rate_chat_pct"] = reopen_rate_chat
    if reopen_rate_chat != 0.0:
        findings["email_vs_chat_reopen_change_pct"] = (reopen_rate_email - reopen_rate_chat) / reopen_rate_chat * 100.0
    else:
        findings["email_vs_chat_reopen_change_pct"] = 0.0

    # q5: High-cost low-satisfaction segment
    seg_mask = (df["labor_cost"] >= 100) & (df["satisfaction_score"].notna()) & (df["satisfaction_score"] <= 2)
    seg = df.loc[seg_mask]
    seg_count = int(len(seg))
    findings["high_cost_low_sat_count"] = seg_count
    if seg_count > 0:
        findings["high_cost_low_sat_reopen_pct"] = float(seg["reopened"].mean()) * 100.0
    else:
        findings["high_cost_low_sat_reopen_pct"] = 0.0

    # q6: Distribution
    findings["resolution_median"] = float(df["resolution_minutes"].median())
    findings["resolution_std"] = float(df["resolution_minutes"].std(ddof=1))
    sat = df["satisfaction_score"].dropna()
    findings["satisfaction_mean"] = float(sat.mean())
    findings["satisfaction_missing_count"] = int(df["satisfaction_score"].isna().sum())

    # q7: Agent concentration
    findings["unique_agent_count"] = int(df["agent_id"].nunique())
    agent_counts = df["agent_id"].value_counts()
    top_count = int(agent_counts.max())
    findings["top_agent_ticket_count"] = top_count
    findings["top_agent_ticket_share_pct"] = top_count / ticket_count * 100.0

    # Ensure all values are finite numbers
    for key, value in findings.items():
        if isinstance(value, (int, np.integer)):
            findings[key] = int(value)
        else:
            v = float(value)
            if not np.isfinite(v):
                v = 0.0
            findings[key] = v

    with open(output_path, "w") as f:
        json.dump(findings, f, indent=2)


if __name__ == "__main__":
    main()
