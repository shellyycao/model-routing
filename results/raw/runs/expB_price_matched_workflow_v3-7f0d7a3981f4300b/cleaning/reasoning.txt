import os
import json
from datetime import datetime

import numpy as np
import pandas as pd

INPUT_DIR = os.environ['INPUT_DIR']
OUTPUT_DIR = os.environ['OUTPUT_DIR']
os.makedirs(OUTPUT_DIR, exist_ok=True)

MISSING_TOKENS = {'', 'na', 'n/a', 'null', 'none', '-'}
DATE_FORMATS = ['%Y-%m-%d', '%m/%d/%Y', '%d-%b-%Y', '%Y/%m/%d', '%B %d, %Y']
BUILDING_MAP = {
    'residential': 'Residential', 'res': 'Residential', 'resi': 'Residential',
    'commercial': 'Commercial', 'comm': 'Commercial',
    'industrial': 'Industrial', 'ind': 'Industrial',
}
TARIFF_MAP = {'peak': 'peak', 'off-peak': 'off-peak', 'offpeak': 'off-peak', 'off peak': 'off-peak'}


def normalize(value):
    if value is None:
        return np.nan
    s = str(value).strip()
    return np.nan if s.lower() in MISSING_TOKENS else s


def parse_date(value):
    if not isinstance(value, str):
        return np.nan
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(value, fmt).strftime('%Y-%m-%d')
        except ValueError:
            continue
    return np.nan


def parse_energy(value):
    if not isinstance(value, str):
        return np.nan
    t = value.replace(',', '').strip()
    low = t.lower()
    factor = 1.0
    if low.endswith('mwh'):
        t = t[:-3].strip()
        factor = 1000.0
    elif low.endswith('kwh'):
        t = t[:-3].strip()
    try:
        return float(t) * factor
    except ValueError:
        return np.nan


def parse_temperature(value):
    if not isinstance(value, str):
        return np.nan
    t = value.strip()
    low = t.lower()
    is_fahrenheit = False
    if low.endswith('f'):
        is_fahrenheit = True
        t = t[:-1]
    elif low.endswith('c'):
        t = t[:-1]
    t = t.strip()
    if t.endswith('\u00b0'):
        t = t[:-1].strip()
    try:
        v = float(t)
    except ValueError:
        return np.nan
    if is_fahrenheit:
        v = (v - 32.0) * 5.0 / 9.0
    if v < -40.0 or v > 50.0:
        return np.nan
    return round(v, 1)

def main():
    raw_path = os.path.join(INPUT_DIR, 'raw.csv')
    df = pd.read_csv(raw_path, dtype=str, keep_default_na=False, na_filter=False)
    raw_rows = int(len(df))

    # R1: strip whitespace and normalise missing tokens
    for col in df.columns:
        df[col] = df[col].map(normalize)

    # R2: drop duplicate reading_id, keep first in raw order

    mask = df['reading_date'].notna()
