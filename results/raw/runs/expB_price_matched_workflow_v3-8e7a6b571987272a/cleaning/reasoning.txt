```python
import os
import json
import re
from datetime import datetime

import pandas as pd
import numpy as np

MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

DATE_FORMATS = ["%Y-%m-%d", "%m/%d/%Y", "%d-%b-%Y", "%Y/%m/%d", "%B %d, %Y"]

BUILDING_MAP = {
    "residential": "Residential",
    "res": "Residential",
    "resi": "Residential",
    "commercial": "Commercial",
    "comm": "Commercial",
    "industrial": "Industrial",
    "ind": "Industrial",
}

TARIFF_MAP = {
    "peak": "peak",
    "off-peak": "off-peak",
    "offpeak": "off-peak",
    "off peak": "off-peak",
}

def parse_date(value):
    if value is None:
        return None
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(value, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None

def parse_energy(value):
    """Return energy in kWh or None if invalid."""
    if value is None:
        return None
    s = value.replace(",", "").strip()
    low = s.lower()
    factor = 1.0
    if low.endswith("mwh"):
        s = s[:-3].strip()
        factor = 1000.0
    elif low.endswith("kwh"):
        s = s[:-3].strip()
    if s == "":
        return None
    try:
        num = float(s)
    except ValueError:
        return None
    kwh = num * factor
    if kwh < 0 or kwh > 100000:
        return None
    return round(kwh, 2)

def parse_temperature(value):
    """Return Celsius (rounded 1dp) or None -> kept as missing."""
    if value is None:
        return None
    s = value.strip()
    low = s.lower()
    unit = "c"
    if low.endswith("°f"):
        s = s[:-2].strip(); unit = "f"
    elif low.endswith("f"):
        s = s[:-1].strip(); unit = "f"
    elif low.endswith("°c"):
        s = s[:-2].strip()
    elif low.endswith("c"):
        s = s[:-1].strip()
    elif low.endswith("°"):
        s = s[:-1].strip()
    if s == "":
        return None
    try:
        num = float(s)
    except ValueError:
        return None
    if unit == "f":
        num = (num - 32) * 5 / 9
    if num < -40 or num > 50:
        return None
    return round(num, 1)
```
