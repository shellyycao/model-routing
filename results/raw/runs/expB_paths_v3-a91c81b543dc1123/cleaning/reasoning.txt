```
def try_parse(val):
    if pd.isna(val):
        return pd.NaT
    val = val.strip()
    # normalize month abbreviation for %d-%b-%Y
    val = re.sub(r'([a-z]{3})', lambda m: m.group(1).title(), val, flags=re.IGNORECASE)
    for fmt in ["%Y-%m-%d", "%m/%d/%Y", "%d-%b-%Y", "%Y/%m/%d", "%B %d, %Y"]:
        try:
            return datetime.strptime(val, fmt).date()
        except:
            continue
    return pd.NaT
```
