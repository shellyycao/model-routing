def parse_date(s):
    if pd.isna(s):
        return None
    s = s.strip()
    for fmt in formats:
        try:
            dt = datetime.datetime.strptime(s, fmt)
            return dt.strftime('%Y-%m-%d')
        except ValueError:
            continue
    return None

def convert_energy(val):
    if pd.isna(val):
        return None
    s = str(val).replace(',', '').strip()
    # detect unit
    pattern = re.compile(r'(?i)^\s*([0-9]*\.?[0-9]+)\s*(mwh|kwh)?\s*$')
    m = pattern.match(s)
    if not m:
        return None
    number = float(m.group(1))
    unit = m.group(2)
    if unit:
        unit = unit.lower()
    if unit == 'mwh':
        number *= 1000
    # if unit is kwh or None, keep as is
    if number < 0 or number > 100000:
        return None
    return round(number, 2)

def convert_temp(val):
    if pd.isna(val):
        return np.nan
    s = str(val).strip()
    # remove degree sign
    s = s.replace('°', '')
    pattern = re.compile(r'(?i)^\s*([+-]?[0-9]*\.?[0-9]+)\s*([cf])?\s*$')
    m = pattern.match(s)
    if not m:
        return np.nan
    number = float(m.group(1))
    unit = m.group(2)
    if unit:
        unit = unit.lower()
    if unit == 'f':
        c = (number - 32) * 5.0 / 9.0
    else:
        c = number
    if c < -40 or c > 50:
        return np.nan
    return round(c, 1)
