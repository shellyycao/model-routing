```
date_formats = [
    "%Y-%m-%d",
    "%m/%d/%Y",
    "%d-%b-%Y",
    "%Y/%m/%d",
    "%B %d, %Y"
]

def parse_date(val):
    if pd.isna(val):
        return None
    for fmt in date_formats:
        try:
            dt = datetime.strptime(val, fmt)
            return dt.strftime('%Y-%m-%d')
        except ValueError:
            continue
    return None

df['reading_date_parsed'] = df['reading_date'].apply(parse_date)
date_invalid = df['reading_date_parsed'].isna().sum()
df = df[~df['reading_date_parsed'].isna()]
df = df.drop(columns=['reading_date'])
df = df.rename(columns={'reading_date_parsed':'reading_date'})
```

```
def parse_energy(val):
    if pd.isna(val):
        return np.nan
    s = str(val).replace(',', '').strip()
    # regex to capture number and optional unit
    m = re.match(r'^([-+]?[0-9]*\.?[0-9]+)\s*([a-zA-Z]*)$', s, re.I)
    if not m:
        return np.nan
    num = float(m.group(1))
    unit = m.group(2).lower()
    if unit == 'mwh':
        num *= 1000
    elif unit == 'kwh' or unit == '':
        pass
    else:
        # unknown unit
        return np.nan
    return round(num, 2)

df['energy_kwh'] = df['energy'].apply(parse_energy)
energy_invalid = df['energy_kwh'].isna().sum()
```

```
def parse_temp(val):
    if pd.isna(val):
        return np.nan
    s = str(val).strip()
    # remove degree sign
    s = s.replace('°', '')
    # regex number + optional unit
    m = re.match(r'^([-+]?[0-9]*\.?[0-9]+)\s*([cCfF]?)$', s)
    if not m:
        return np.nan
    num = float(m.group(1))
    unit = m.group(2).lower()
    if unit == 'f':
        c = (num - 32) * 5.0 / 9.0
    else:  # C or empty
        c = num
    c = round(c, 1)
    if c < -40 or c > 50:
        return np.nan
    return c

df['temperature_c'] = df['temperature'].apply(parse_temp)
temperature_missing = df['temperature_c'].isna().sum()
```
