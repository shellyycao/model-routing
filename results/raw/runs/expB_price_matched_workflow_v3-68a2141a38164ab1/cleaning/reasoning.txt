```python
DATE_FORMATS = ["%Y-%m-%d", "%m/%d/%Y", "%d-%b-%Y", "%Y/%m/%d", "%B %d, %Y"]

def parse_date(v):
    if v is None or (isinstance(v, float) and np.isnan(v)):
        return None
    s = str(v).strip()
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None
```

```python
import json
import math

import pandas as pd

    "%d-%b-%Y",   # DD-Mon-YYYY

    "apparel": "Clothing",
    "home & garden": "Home & Garden",

    "home&garden": "Home & Garden",
    "sports": "Sports",
    "sport": "Sports",
    "books": "Books",
    "book": "Books",
}

summary = {}

def is_missing_value(v):
    return v is None or (isinstance(v, float) and math.isnan(v)) or str(v).strip().lower() in MISSING_TOKENS

def parse_order_date(v):
    if v is None or (isinstance(v, float) and math.isnan(v)):
        return None
    s = str(v).strip()
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt).strftime("%Y-%m-%d")
        except ValueError:
            pass
    return None

def map_region(v):
    if v is None or (isinstance(v, float) and math.isnan(v)):
        return None
    key = " ".join(str(v).strip().lower().split())
    return REGION_MAP.get(key)

def map_category(v):
    if v is None or (isinstance(v, float) and math.isnan(v)):
        return None
    key = " ".join(str(v).strip().lower().split())
    return CATEGORY_MAP.get(key)

def parse_unit_price(v):
    if v is None or (isinstance(v, float) and math.isnan(v)):
        return None
    s = str(v).strip()
    s = s.replace("$", "").replace(",", "")
    s = re.sub(r"(?i)usd", "", s)
    s = s.strip()
    if not s:
        return None
    try:
        x = float(s)
    except ValueError:
        return None
    if math.isnan(x) or math.isinf(x):
        return None
    if x <= 0 or x > 5000:
        return None
    return x

def parse_quantity(v):
    if v is None or (isinstance(v, float) and math.isnan(v)):
        return None
    s = str(v).strip()
    try:
        x = float(s)
    except ValueError:
        return None
    if math.isnan(x) or math.isinf(x) or not x.is_integer():
        return None
    q = int(x)
    if q < 1 or q > 50:
        return None
    return q

def parse_discount(v):

    if v is None or (isinstance(v, float) and math.isnan(v)):
        return 0.0
    s = str(v).strip()
    if s.endswith("%"):

    if not s:
        return None
    try:
        x = float(s)
    except ValueError:
        return None
    if math.isnan(x) or math.isinf(x):
        return None
    if x < 0 or x > 100:
        return None
    return x
```

```python
import os
import re
import json
import math
from datetime import datetime

import numpy as np
import pandas as pd

INPUT_DIR = os.environ["INPUT_DIR"]
OUTPUT_DIR = os.environ["OUTPUT_DIR"]

MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

DATE_FORMATS = (
    "%Y-%m-%d",   # YYYY-MM-DD
    "%m/%d/%Y",   # MM/DD/YYYY (month first)
    "%d-%b-%Y",   # DD-Mon-YYYY, English 3-letter month
    "%Y/%m/%d",   # YYYY/MM/DD
    "%B %d, %Y",  # Month D, YYYY, full English month name
)

REGION_MAP = {
    "north": "North", "south": "South", "east": "East", "west": "West",
    "n": "North", "s": "South", "e": "East", "w": "West",
}

CATEGORY_MAP = {
    "electronics": "Electronics", "electronic": "Electronics",
    "clothing": "Clothing", "apparel": "Clothing",
    "home & garden": "Home & Garden", "home and garden": "Home & Garden",
    "home&garden": "Home & Garden",
    "sports": "Sports", "sport": "Sports",
    "books": "Books", "book": "Books",
}

OUTPUT_COLUMNS = ["order_id", "order_date", "customer_id", "region",
                  "product_category", "quantity", "unit_price",
                  "discount_pct", "revenue"]


def _is_nan(v):
    return isinstance(v, float) and math.isnan(v)


def _norm(v):
    """Lower-cased, whitespace-collapsed key for lookups."""
    return " ".join(str(v).strip().lower().split())


def parse_order_date(v):
    if v is None or _is_nan(v):
        return None
    s = str(v).strip()
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(s, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None


def map_region(v):
    if v is None or _is_nan(v):
        return None
    return REGION_MAP.get(_norm(v))


def map_category(v):
    if v is None or _is_nan(v):
        return None
    return CATEGORY_MAP.get(_norm(v))


def parse_unit_price(v):
    if v is None or _is_nan(v):
        return None
    s = str(v)
    s = s.replace("$", "").replace(",", "")
    s = re.sub(r"(?i)usd", "", s).strip()
    if not s:
        return None
    try:
        x = float(s)
    except ValueError:
        return None
    if math.isnan(x) or math.isinf(x) or x <= 0 or x > 5000:
        return None
    return x


def parse_quantity(v):
    if v is None or _is_nan(v):
        return None
    try:
        x = float(str(v).strip())
    except ValueError:
        return None
    if math.isnan(x) or math.isinf(x) or not x.is_integer():
        return None
    q = int(x)
    if q < 1 or q > 50:
        return None
    return q


def parse_discount(v):
    # None -> drop the row; a missing value means "no discount" -> 0.0
    if v is None or _is_nan(v):
        return 0.0
    s = str(v).strip()
    if s.endswith("%"):
        s = s[:-1].strip()
    if not s:
        return None
    try:
        x = float(s)
    except ValueError:
        return None
    if math.isnan(x) or math.isinf(x) or x < 0 or x > 100:
        return None
    return x


def apply_rule(df, column, func, summary, key):
    mapped = df[column].map(func)
    keep = mapped.notna()
    summary[key] = int((~keep).sum())
    out = df.loc[keep].copy()
    out[column] = mapped[keep]
    return out


def main():
    summary = {}

    raw_path = os.path.join(INPUT_DIR, "raw.csv")
    df = pd.read_csv(raw_path, dtype=str, keep_default_na=False, na_filter=False)
    summary["raw_rows"] = int(len(df))

    # R1: strip whitespace; normalize missing tokens to NaN
    df = df.apply(lambda col: col.astype(str).str.strip())
    for col in df.columns:
        df[col] = df[col].map(lambda v: np.nan if v.lower() in MISSING_TOKENS else v)

    # R2: drop duplicate order_id, keep first occurrence in raw-file order
    before = len(df)
    df = df.drop_duplicates(subset="order_id", keep="first")
    summary["r2_duplicate_order_id_rows_removed"] = int(before - len(df))

    # order_id must not be null in the output schema
    before = len(df)
    df = df[df["order_id"].notna()]
    summary["missing_order_id_rows_dropped"] = int(before - len(df))

    # R3-R8
    df = apply_rule(df, "order_date", parse_order_date, summary, "r3_invalid_order_date_rows_dropped")
    df = apply_rule(df, "region", map_region, summary, "r4_invalid_region_rows_dropped")
    df = apply_rule(df, "product_category", map_category, summary, "r5_invalid_product_category_rows_dropped")
    df = apply_rule(df, "unit_price", parse_unit_price, summary, "r6_invalid_unit_price_rows_dropped")
    df = apply_rule(df, "quantity", parse_quantity, summary, "r7_invalid_quantity_rows_dropped")
    df = apply_rule(df, "discount_pct", parse_discount, summary, "r8_invalid_discount_pct_rows_dropped")

    # R9: customer_id upper-case; missing -> 'UNKNOWN'
    missing_cust = df["customer_id"].isna()
    summary["r9_missing_customer_id_filled_unknown"] = int(missing_cust.sum())
    df["customer_id"] = df["customer_id"].astype(str).str.strip().str.upper()
    df.loc[missing_cust, "customer_id"] = "UNKNOWN"
```
