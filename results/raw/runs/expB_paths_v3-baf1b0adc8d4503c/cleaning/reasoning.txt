def clean_missing(df):
    df = df.applymap(lambda x: x.strip() if isinstance(x, str) else x)
    missing_vals = {'', 'NA', 'N/A', 'null', 'None', '-'}
    df = df.replace(missing_vals, np.nan)
    return df

def parse_hourly_cost(val):
    if pd.isna(val):
        return np.nan
    s = str(val).replace('$','').replace('USD','').replace(',','').strip()
    try:
        return float(s)
    except:
        return np.nan

Team mapping:

def map_team(val):
    if pd.isna(val):
        return np.nan
    low = val.lower()
    if 'alpha' in low:
        return 'Alpha'
    if 'beta' in low:
        return 'Beta'
    if 'gamma' in low:
        return 'Gamma'
    return np.nan

def map_region(val):
    if pd.isna(val):
        return np.nan
    low = val.lower()
    if low in ('east','e'):
        return 'East'
    if low in ('west','w'):
        return 'West'
    return np.nan
