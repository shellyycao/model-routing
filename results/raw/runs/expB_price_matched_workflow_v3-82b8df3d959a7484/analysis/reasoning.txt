```python
import json
import os

import numpy as np
import pandas as pd


def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")
    output_path = os.path.join(os.environ["OUTPUT_DIR"], "findings.json")

    df = pd.read_csv(input_path)

    # Ensure numeric types
    df["revenue"] = pd.to_numeric(df["revenue"], errors="coerce")
    df["discount_pct"] = pd.to_numeric(df["discount_pct"], errors="coerce")
    df["order_date"] = pd.to_datetime(df["order_date"], errors="coerce")

    df = df.dropna(subset=["revenue", "order_date"])  # should be no-op on cleaned data

    findings = {}

    # q1
    total_revenue = float(df["revenue"].sum())
    order_count = int(len(df))
    findings["total_revenue"] = total_revenue
    findings["order_count"] = float(order_count)
    findings["average_order_value"] = total_revenue / order_count if order_count else 0.0

    # q2
    by_region = df.groupby("region")["revenue"].sum()
    for region in ["North", "South", "East", "West"]:
        findings[f"revenue_{region.lower()}"] = float(by_region.get(region, 0.0))

    # q3
    df["year"] = df["order_date"].dt.year
    df["quarter"] = df["order_date"].dt.quarter
    df24 = df[df["year"] == 2024]
    by_q = df24.groupby("quarter")["revenue"].sum()
    for q in [1, 2, 3, 4]:
        findings[f"revenue_2024_q{q}"] = float(by_q.get(q, 0.0))

    # q4
    revenue_2023 = float(df.loc[df["year"] == 2023, "revenue"].sum())
    revenue_2024 = float(df.loc[df["year"] == 2024, "revenue"].sum())
    findings["revenue_2023"] = revenue_2023
    findings["revenue_2024"] = revenue_2024
    findings["revenue_yoy_change_pct"] = ((revenue_2024 - revenue_2023) / revenue_2023 * 100) if revenue_2023 else 0.0

    # q5
    mask = (df["product_category"] == "Electronics") & (df["discount_pct"] >= 20)
    findings["electronics_high_discount_orders"] = float(int(mask.sum()))
    findings["electronics_high_discount_revenue"] = float(df.loc[mask, "revenue"].sum())

    # q6
    rev = df["revenue"]
    findings["order_revenue_mean"] = float(rev.mean())
    findings["order_revenue_median"] = float(rev.median())
    findings["order_revenue_std"] = float(rev.std(ddof=1))
    findings["order_revenue_min"] = float(rev.min())
    findings["order_revenue_max"] = float(rev.max())

    # q7
    by_cust = df.groupby("customer_id")["revenue"].sum()
    findings["unique_customer_count"] = float(df["customer_id"].nunique())
    top = float(by_cust.max()) if len(by_cust) else 0.0
    findings["top_customer_revenue"] = top
    findings["top_customer_revenue_share_pct"] = (top / total_revenue * 100) if total_revenue else 0.0

    # sanitize
    for k, v in findings.items():
        if not np.isfinite(v):
            findings[k] = 0.0

    with open(output_path, "w") as f:
        json.dump(findings, f, indent=2)


if __name__ == "__main__":
    main()
```
