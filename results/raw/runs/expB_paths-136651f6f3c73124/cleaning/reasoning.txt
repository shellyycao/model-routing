```
import os, json, pandas as pd, numpy as np, re
from datetime import datetime

input_path = os.path.join(os.environ['INPUT_DIR'], 'raw.csv')
output_path = os.path.join(os.environ['OUTPUT_DIR'], 'cleaned.csv')
summary_path = os.path.join(os.environ['OUTPUT_DIR'], 'cleaning_summary.json')

missing_vals = {'', 'na', 'n/a', 'null', 'none', '-'}
df = pd.read_csv(input_path, dtype=str, keep_default_na=False)

# strip whitespace
df = df.applymap(lambda x: x.strip() if isinstance(x, str) else x)

# normalize missing
def to_nan(v):
    if isinstance(v, str) and v.lower() in missing_vals:
        return np.nan
    return v
df = df.applymap(to_nan)

summary = {'raw_rows': len(df)}
# R2 dedup
dup_before = len(df)
df = df.drop_duplicates(subset='order_id', keep='first')
summary['duplicates_removed'] = dup_before - len(df)

# R3 order_date
def parse_date(val):
    if pd.isna(val):
        return np.nan
    val = val.strip()
    formats = ['%Y-%m-%d','%m/%d/%Y','%d-%b-%Y','%Y/%m/%d','%B %d, %Y']
    for fmt in formats:
        try:
            dt = datetime.strptime(val, fmt)
            return dt.strftime('%Y-%m-%d')
        except Exception:
            continue
    return np.nan

df['order_date'] = df['order_date'].apply(parse_date)
summary['order_date_dropped'] = df['order_date'].isna().sum()
df = df.dropna(subset=['order_date'])

# R4 region
region_map = {
    'n':'North','north':'North',
    's':'South','south':'South',
    'e':'East','east':'East',
    'w':'West','west':'West'
}
def map_region(val):
    if pd.isna(val):
        return np.nan
    key = val.strip().lower()
    return region_map.get(key, np.nan)
df['region'] = df['region'].apply(map_region)
summary['region_dropped'] = df['region'].isna().sum()
df = df.dropna(subset=['region'])

# R5 product_category
cat_map = {
    'electronics':'Electronics','electronic':'Electronics',
    'clothing':'Clothing','apparel':'Clothing',
    'home & garden':'Home & Garden','home and garden':'Home & Garden','home&garden':'Home & Garden',
    'sports':'Sports','sport':'Sports',
    'books':'Books','book':'Books'
}
def map_category(val):
    if pd.isna(val):
        return np.nan
    key = val.strip().lower()
    return cat_map.get(key, np.nan)
df['product_category'] = df['product_category'].apply(map_category)
summary['product_category_dropped'] = df['product_category'].isna().sum()
df = df.dropna(subset=['product_category'])

# R6 unit_price
price_pattern = re.compile(r'\$|USD|,', flags=re.IGNORECASE)
def clean_price(val):
    if pd.isna(val):
        return np.nan
    s = price_pattern.sub('', str(val)).strip()
    if s == '':
        return np.nan
    try:
        p = float(s)
        if p <= 0 or p > 5000:
            return np.nan
        return p
    except:
        return np.nan
df['unit_price'] = df['unit_price'].apply(clean_price)
summary['unit_price_dropped'] = df['unit_price'].isna().sum()
df = df.dropna(subset=['unit_price'])

# R7 quantity
def clean_quantity(val):
    if pd.isna(val):
        return np.nan
    try:
        q = float(val)
        if not q.is_integer():
            return np.nan
        q_int = int(q)
        if q_int < 1 or q_int > 50:
            return np.nan
        return q_int
    except:
        return np.nan
df['quantity'] = df['quantity'].apply(clean_quantity)
summary['quantity_dropped'] = df['quantity'].isna().sum()
df = df.dropna(subset=['quantity'])

# R8 discount_pct
def clean_discount(val):
    if pd.isna(val):
        return 0.0
    s = str(val).strip()
    if s.endswith('%'):
        s = s[:-1].strip()
    try:
        d = float(s)
        if d < 0 or d > 100:
            return np.nan
        return d
    except:
        return np.nan
df['discount_pct'] = df['discount_pct'].apply(clean_discount)
summary['discount_pct_dropped'] = df['discount_pct'].isna().sum()
df = df.dropna(subset=['discount_pct'])

# R9 customer_id
def clean_customer(val):
    if pd.isna(val):
        return 'UNKNOWN'
    return str(val).strip().upper()
df['customer_id'] = df['customer_id'].apply(clean_customer)

# R10 revenue
df['revenue'] = (df['quantity'] * df['unit_price'] * (1 - df['discount_pct']/100)).round(2)

# Ensure order_id not missing
summary['order_id_missing'] = df['order_id'].isna().sum()
df = df.dropna(subset=['order_id'])

# Select and order columns
final_cols = ['order_id','order_date','customer_id','region','product_category',
              'quantity','unit_price','discount_pct','revenue']
df = df[final_cols]

# Sort by order_id ascending
df = df.sort_values('order_id')

# Write cleaned CSV
df.to_csv(output_path, index=False)

summary['cleaned_rows'] = len(df)
# write summary
with open(summary_path, 'w') as f:
    json.dump(summary, f, indent=2)
```
