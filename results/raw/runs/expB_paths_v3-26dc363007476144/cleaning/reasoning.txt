        return 'Alpha'

def parse_cost(val):
    if pd.isna(val):
        return np.nan
    cleaned = re.sub(r'[\$,]', '', val, flags=re.IGNORECASE)
    cleaned = cleaned.replace('USD', '').strip()
    try:
        cost = float(cleaned)
        return cost
    except:
        return np.nan

def parse_resolution(val):
    if pd.isna(val):
        return np.nan
    # remove non-digit characters
    m = re.search(r'(\d+)', val)
    if m:
        minutes = int(m.group(1))
        if 1 <= minutes <= 480:
            return minutes
    return np.nan
