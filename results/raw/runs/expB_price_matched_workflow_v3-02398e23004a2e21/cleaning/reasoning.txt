```python
import os, re, json
import pandas as pd
import numpy as np

INPUT_DIR = os.environ["INPUT_DIR"]
OUTPUT_DIR = os.environ["OUTPUT_DIR"]

MISSING = {"", "na", "n/a", "null", "none", "-"}

def read_csv_text(path):
    df = pd.read_csv(path, dtype=str, keep_default_na=False)
    df.columns = [c.strip() for c in df.columns]
    for c in df.columns:
        df[c] = df[c].astype(str).str.strip()
        df[c] = df[c].where(~df[c].str.lower().isin(MISSING), pd.NA)
    return df

summary = {}

tickets = read_csv_text(os.path.join(INPUT_DIR, "tickets.csv"))
agents = read_csv_text(os.path.join(INPUT_DIR, "agents.csv"))
```

```python
TEAM_MAP = {"alpha": "Alpha", "team alpha": "Alpha", "beta": "Beta", "team beta": "Beta", "gamma": "Gamma", "team gamma": "Gamma"}
REGION_MAP = {"east": "East", "e": "East", "west": "West", "w": "West"}

def parse_cost(s):
    if pd.isna(s): return None
    t = s.replace("$", "").replace(",", "")
    t = re.sub(r"(?i)usd", "", t).strip()
    try:
        v = float(t)
    except ValueError:
        return None
    return v if v > 0 else None

agents["team"] = agents["team"].str.lower().map(TEAM_MAP)
agents["region"] = agents["region"].str.lower().map(REGION_MAP)
agents["hourly_cost"] = agents["hourly_cost"].map(parse_cost)
before = len(agents)
agents = agents.dropna(subset=["team","region","hourly_cost"])
summary["r4_roster_rows_excluded"] = int(before - len(agents))
```
