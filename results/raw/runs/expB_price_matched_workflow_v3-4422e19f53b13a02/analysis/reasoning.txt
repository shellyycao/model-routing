```python
import json
import os

import pandas as pd


def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")
    output_path = os.path.join(os.environ["OUTPUT_DIR"], "findings.json")

    df = pd.read_csv(input_path)

    findings = {}

    ticket_count = int(len(df))
    findings["ticket_count"] = float(ticket_count)
    findings["mean_resolution_minutes"] = float(df["resolution_minutes"].mean())
    findings["total_labor_cost"] = float(df["labor_cost"].sum())

    team_cost = df.groupby("team")["labor_cost"].sum()
    for team, key in (("Alpha", "labor_cost_alpha"), ("Beta", "labor_cost_beta"), ("Gamma", "labor_cost_gamma")):
        findings[key] = float(team_cost.get(team, 0.0))

    chan_res = df.groupby("channel")["resolution_minutes"].mean()
    for ch, key in (("email", "resolution_email"), ("chat", "resolution_chat"), ("phone", "resolution_phone")):
        findings[key] = float(chan_res.get(ch, 0.0))

    def reopen_pct(sub):
        if len(sub) == 0:
            return 0.0
        return float(sub["reopened"].mean() * 100.0)

    email = df[df["channel"] == "email"]
    chat = df[df["channel"] == "chat"]
    email_pct = reopen_pct(email)
    chat_pct = reopen_pct(chat)
    findings["reopen_rate_email_pct"] = email_pct
    findings["reopen_rate_chat_pct"] = chat_pct
    if chat_pct != 0.0:
        findings["email_vs_chat_reopen_change_pct"] = float((email_pct - chat_pct) / chat_pct * 100.0)
    else:
        findings["email_vs_chat_reopen_change_pct"] = 0.0

    sat = df["satisfaction_score"]
    seg = df[(df["labor_cost"] >= 100) & sat.notna() & (sat <= 2)]
    seg_count = int(len(seg))
    findings["high_cost_low_sat_count"] = float(seg_count)
    findings["high_cost_low_sat_reopen_pct"] = float(seg["reopened"].mean() * 100.0) if seg_count > 0 else 0.0

    findings["resolution_median"] = float(df["resolution_minutes"].median())
    findings["resolution_std"] = float(df["resolution_minutes"].std(ddof=1))
    findings["satisfaction_mean"] = float(sat.mean())  # skipna default
    findings["satisfaction_missing_count"] = float(int(sat.isna().sum()))

    findings["unique_agent_count"] = float(int(df["agent_id"].nunique()))
    top_count = int(df["agent_id"].value_counts().max())
    findings["top_agent_ticket_count"] = float(top_count)
    findings["top_agent_ticket_share_pct"] = float(top_count / ticket_count * 100.0) if ticket_count > 0 else 0.0

    os.makedirs(os.environ["OUTPUT_DIR"], exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(findings, f, indent=2)


if __name__ == "__main__":
    main()
```
