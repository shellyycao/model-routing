}

    "yes": 1, "y": 1, "true": 1, "1": 1,
    "no": 0, "n": 0, "false": 0, "0": 0,
}
RES_SUFFIX_RE = re.compile(r"(?i)\s*(?:minutes|min)\s*$")
INT_RE = re.compile(r"\d+")


def read_text_csv(path):
    """R1: read all columns as text, strip whitespace, normalise missing tokens."""
    frame = pd.read_csv(path, dtype=str, keep_default_na=False)
    frame.columns = [str(c).strip() for c in frame.columns]
    for col in frame.columns:
        values = frame[col].astype(str).str.strip()
        frame[col] = values.where(~values.str.lower().isin(MISSING_TOKENS), pd.NA)
    return frame


def parse_hourly_cost(value):
    """R4: strip $ / USD / commas and parse a positive float."""
    if pd.isna(value):
        return None
    text = re.sub(r"(?i)usd", "", str(value)).replace("$", "").replace(",", "").strip()
    if not text:
        return None
    try:
        return float(text)
    except ValueError:
        return None


def parse_resolution_minutes(value):
    """R7: drop trailing 'min'/'minutes', parse a whole number in [1, 480]."""
    if pd.isna(value):
        return None
    text = RES_SUFFIX_RE.sub("", str(value)).strip()
    if not INT_RE.fullmatch(text):
        return None
    minutes = int(text)
    return minutes if 1 <= minutes <= 480 else None


def parse_satisfaction(value):
    """R8: whole number in [1, 5]; anything else becomes missing (row kept)."""
    if pd.isna(value):
        return pd.NA
    text = str(value).strip()
    if INT_RE.fullmatch(text):
        score = int(text)
        if 1 <= score <= 5:
            return score
    return pd.NA
