def clean_strings(df):
    for col in df.columns:
        df[col] = df[col].astype(str).str.strip()
    # replace missing tokens with np.nan
    missing_vals = ['', 'NA', 'N/A', 'null', 'None', '-']
    df.replace(missing_vals, np.nan, inplace=True)
    return df
