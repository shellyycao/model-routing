df = pd.read_csv(..., dtype=str, keep_default_na=False)

raw_rows = len(df)

# strip whitespace
df = df.applymap(lambda x: x.strip() if isinstance(x, str) else x)

# treat missing values
missing_vals = {'', 'na', 'n/a', 'null', 'none', '-'}
df = df.replace(missing_vals, np.nan, regex=False)

# duplicate removal
dup_mask = df.duplicated(subset='order_id', keep='first')
duplicates_removed = dup_mask.sum()

# Now start filters, keep track counts
def parse_date(val):
    if pd.isna(val):
        return pd.NaT
    for fmt in ["%Y-%m-%d","%m/%d/%Y","%d-%b-%Y","%Y/%m/%d","%B %d, %Y"]:
        try:
            return datetime.strptime(val, fmt).date()
        except Exception:
            continue
    return pd.NaT

df['order_date_parsed'] = df['order_date'].apply(parse_date)
date_drop = df['order_date_parsed'].isna().sum()
df = df[~df['order_date_parsed'].isna()]
df['order_date'] = df['order_date_parsed'].astype(str)  # will be YYYY-MM-DD

def clean_price(val):
    if pd.isna(val):
        return np.nan
    s = val.replace('$','').replace('USD','').replace(',','')
    s = s.strip()
    if s == '':
        return np.nan
    try:
        return float(s)
    except:
        return np.nan
df['unit_price_clean'] = df['unit_price'].apply(clean_price)
price_invalid = df['unit_price_clean'].isna() | (df['unit_price_clean']<=0) | (df['unit_price_clean']>5000)
price_drop = price_invalid.sum()
df = df[~price_invalid]
df['unit_price'] = df['unit_price_clean']
df = df.drop(columns=['unit_price_clean'])

# quantity
def parse_qty(val):
    if pd.isna(val):
        return np.nan
    try:
        f = float(val)
        return f
    except:
        return np.nan
df['quantity_num'] = df['quantity'].apply(parse_qty)
qty_invalid = df['quantity_num'].isna() | (df['quantity_num']%1 !=0) | (df['quantity_num']<1) | (df['quantity_num']>50)

# discount_pct
def parse_discount(val):
    if pd.isna(val):
        return 0.0
    s = str(val).strip()
    if s.endswith('%'):
        s = s[:-1]
    s = s.strip()
    if s == '':
        return 0.0
    try:
        return float(s)
    except:
        return np.nan
df['discount_pct_clean'] = df['discount_pct'].apply(parse_discount)

df['discount_pct'] = df['discount_pct_clean']
