```python
import os
import json
import math

import pandas as pd
import numpy as np


def _num(value):
    """Convert to a finite Python float (fallback 0.0)."""
    try:
        v = float(value)
    except (TypeError, ValueError):
        return 0.0
    if not math.isfinite(v):
        return 0.0
    return v


def main():
    input_path = os.path.join(os.environ["INPUT_DIR"], "cleaned.csv")
    output_path = os.path.join(os.environ["OUTPUT_DIR"], "findings.json")

    df = pd.read_csv(input_path)

    findings = {}

    # q1
    ticket_count = int(len(df))
    findings["ticket_count"] = ticket_count
    findings["mean_resolution_minutes"] = _num(df["resolution_minutes"].mean())
    findings["total_labor_cost"] = _num(df["labor_cost"].sum())

    # q2
    team_cost = df.groupby("team")["labor_cost"].sum()
    for team, key in (("Alpha", "labor_cost_alpha"),
                      ("Beta", "labor_cost_beta"),
                      ("Gamma", "labor_cost_gamma")):
        findings[key] = _num(team_cost.get(team, 0.0))

    # q3
    chan_res = df.groupby("channel")["resolution_minutes"].mean()
    for chan, key in (("email", "resolution_email"),
                      ("chat", "resolution_chat"),
                      ("phone", "resolution_phone")):
        findings[key] = _num(chan_res.get(chan, 0.0))

    # q4
    reopen_by_channel = df.groupby("channel")["reopened"].mean()
    email_rate = float(reopen_by_channel.get("email", 0.0))
    chat_rate = float(reopen_by_channel.get("chat", 0.0))
    findings["reopen_rate_email_pct"] = _num(email_rate * 100)
    findings["reopen_rate_chat_pct"] = _num(chat_rate * 100)
    if chat_rate > 0:
        findings["email_vs_chat_reopen_change_pct"] = _num((email_rate - chat_rate) / chat_rate * 100)
    else:
        findings["email_vs_chat_reopen_change_pct"] = 0.0

    # q5
    seg = df[(df["labor_cost"] >= 100) & (df["satisfaction_score"].notna()) & (df["satisfaction_score"] <= 2)]
    seg_count = int(len(seg))
    findings["high_cost_low_sat_count"] = seg_count
    if seg_count > 0:
        findings["high_cost_low_sat_reopen_pct"] = _num(seg["reopened"].mean() * 100)
    else:
        findings["high_cost_low_sat_reopen_pct"] = 0.0

    # q6
    findings["resolution_median"] = _num(df["resolution_minutes"].median())
    findings["resolution_std"] = _num(df["resolution_minutes"].std(ddof=1))
    findings["satisfaction_mean"] = _num(df["satisfaction_score"].mean())
    findings["satisfaction_missing_count"] = int(df["satisfaction_score"].isna().sum())

    # q7
    agent_counts = df["agent_id"].value_counts()
    unique_agents = int(df["agent_id"].nunique())
    top_count = int(agent_counts.max()) if len(agent_counts) else 0
    findings["unique_agent_count"] = unique_agents
    findings["top_agent_ticket_count"] = top_count
    findings["top_agent_ticket_share_pct"] = _num(top_count / ticket_count * 100) if ticket_count else 0.0

    # Ensure all values are native int/float and finite
    clean = {}
    for k, v in findings.items():
        if isinstance(v, (int, float)):
            if isinstance(v, float) and not math.isfinite(v):
                clean[k] = 0.0
            else:
                clean[k] = v
        else:
            clean[k] = _num(v)

    os.makedirs(os.environ["OUTPUT_DIR"], exist_ok=True)
    with open(output_path, "w") as f:
        json.dump(clean, f)


if __name__ == "__main__":
    main()
```
