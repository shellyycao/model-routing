```
def clean_strings(df):
    for col in df.columns:
        if df[col].dtype == object:
            df[col] = df[col].astype(str).str.strip()
    # replace missing markers
    missing_vals = ["", "NA", "N/A", "null", "None", "-"]
    df.replace(missing_vals, np.nan, inplace=True)
    return df
```

```
def map_team(val):
    if pd.isna(val):
        return np.nan
    v = val.lower()
    if 'alpha' in v:
        return 'Alpha'
    if 'beta' in v:
        return 'Beta'
    if 'gamma' in v:
        return 'Gamma'
    return np.nan
```

Hourly cost parsing:

```
def parse_cost(val):
    if pd.isna(val):
        return np.nan
    s = re.sub(r'[\$,]', '', val, flags=re.IGNORECASE)
    s = s.replace('USD', '').strip()
    try:
        cost = float(s)
        if cost <= 0:
            return np.nan
        return cost
    except:
        return np.nan
```

R7: resolution_minutes.

```
def parse_resolution(val):
    if pd.isna(val):
        return np.nan
    s = val.lower()
    s = s.replace('minutes','').replace('minute','').replace('min','')
    s = s.strip()
    if not s.isdigit():
        return np.nan
    minutes = int(s)
    if 1 <= minutes <= 480:
        return minutes
    return np.nan
tickets['resolution_minutes'] = tickets['resolution_minutes'].apply(parse_resolution)
invalid_resolution = tickets['resolution_minutes'].isna().sum()
tickets = tickets.dropna(subset=['resolution_minutes'])
```
