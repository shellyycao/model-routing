def map_team(val):
    if pd.isna(val):
        return np.nan
    v = val.lower()
    if 'alpha' in v:
        return 'Alpha'
    if 'beta' in v:
        return 'Beta'
    if 'gamma' in v:
        return 'Gamma'
    return np.nan

def parse_cost(val):
    if pd.isna(val):
        return np.nan
    # remove $ and USD and commas
    cleaned = re.sub(r'[\$,]', '', str(val), flags=re.IGNORECASE)
    cleaned = re.sub(r'(?i)usd', '', cleaned)
    cleaned = cleaned.strip()
    try:
        cost = float(cleaned)
        return cost if cost > 0 else np.nan
    except:
        return np.nan

def map_channel(val):
    if pd.isna(val):
        return np.nan
    v = val.lower()
    if v in ['email', 'e-mail']:
        return 'email'
    if v in ['chat', 'live chat']:
        return 'chat'
    if v in ['phone', 'call']:
        return 'phone'
    return np.nan

Resolution parsing:

def parse_resolution(val):
    if pd.isna(val):
        return np.nan
    v = str(val).lower()
    v = v.replace('minutes', '').replace('minute', '').replace('min', '').strip()
    # remove any non-digit characters
    v = re.sub(r'[^0-9]', '', v)
    if v == '':
        return np.nan
    try:
        minutes = int(v)
        if 1 <= minutes <= 480:
            return minutes
        else:
            return np.nan
    except:
        return np.nan

def map_reopened(val):
    if pd.isna(val):
        return np.nan
    v = str(val).strip().lower()
    if v in ['yes', 'y', 'true', '1']:
        return 1
    if v in ['no', 'n', 'false', '0']:
        return 0
    return np.nan
