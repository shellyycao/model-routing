```python
import os
import json
import pandas as pd
import numpy as np

INPUT_DIR = os.environ["INPUT_DIR"]
OUTPUT_DIR = os.environ["OUTPUT_DIR"]

df = pd.read_csv(os.path.join(INPUT_DIR, "cleaned.csv"), dtype={"customer_id": str})

# Parse types
df["signup_date"] = pd.to_datetime(df["signup_date"], format="%Y-%m-%d")
df["monthly_fee"] = pd.to_numeric(df["monthly_fee"], errors="coerce")
df["tenure_months"] = pd.to_numeric(df["tenure_months"], errors="coerce")
df["support_tickets"] = pd.to_numeric(df["support_tickets"], errors="coerce")
df["satisfaction_score"] = pd.to_numeric(df["satisfaction_score"], errors="coerce")
df["churned"] = pd.to_numeric(df["churned"], errors="coerce")
df["lifetime_value"] = pd.to_numeric(df["lifetime_value"], errors="coerce")

def churn_rate(sub):
    if len(sub) == 0:
        return 0.0
    return float(sub["churned"].mean() * 100)

metrics = {}

metrics["customer_count"] = int(len(df))
metrics["churn_rate_pct"] = churn_rate(df)
metrics["avg_monthly_fee"] = float(df["monthly_fee"].mean())

# q2 by plan
for plan, key in [("Basic","churn_rate_basic_pct"), ("Standard","churn_rate_standard_pct"), ("Premium","churn_rate_premium_pct")]:
    metrics[key] = churn_rate(df[df["plan"] == plan])

# q3 cohorts
years = df["signup_date"].dt.year
for y in (2021, 2022, 2023, 2024):
    metrics[f"churn_rate_cohort_{y}_pct"] = churn_rate(df[years == y])

# q4 contract
m = churn_rate(df[df["contract_type"] == "Monthly"])
a = churn_rate(df[df["contract_type"] == "Annual"])
metrics["churn_rate_monthly_pct"] = m
metrics["churn_rate_annual_pct"] = a
metrics["churn_monthly_vs_annual_change_pct"] = (m - a) / a * 100 if a != 0 else 0.0

# q5 at-risk
mask = (df["support_tickets"] >= 5) & (df["satisfaction_score"].notna()) & (df["satisfaction_score"] <= 2)
at_risk = df[mask]
metrics["at_risk_customer_count"] = int(len(at_risk))
metrics["at_risk_churn_rate_pct"] = churn_rate(at_risk)

# q6
metrics["tenure_mean"] = float(df["tenure_months"].mean())
metrics["tenure_median"] = float(df["tenure_months"].median())
metrics["tenure_std"] = float(df["tenure_months"].std(ddof=1))
metrics["satisfaction_mean"] = float(df["satisfaction_score"].mean())
metrics["satisfaction_missing_count"] = int(df["satisfaction_score"].isna().sum())

# q7
ltv_b = float(df.loc[df["plan"]=="Basic", "lifetime_value"].mean())
ltv_s = float(df.loc[df["plan"]=="Standard", "lifetime_value"].mean())
ltv_p = float(df.loc[df["plan"]=="Premium", "lifetime_value"].mean())
metrics["ltv_basic"] = ltv_b
metrics["ltv_standard"] = ltv_s
metrics["ltv_premium"] = ltv_p
metrics["premium_vs_basic_ltv_change_pct"] = (ltv_p - ltv_b) / ltv_b * 100 if ltv_b != 0 else 0.0

# Ensure all finite
for k, v in metrics.items():
    metrics[k] = float(v)
    if not np.isfinite(metrics[k]):
        metrics[k] = 0.0
```
