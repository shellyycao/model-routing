import json
import math
import os
import re
from datetime import datetime

import pandas as pd


MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

DATE_FORMATS = (
    "%Y-%m-%d",    # YYYY-MM-DD
    "%m/%d/%Y",    # MM/DD/YYYY (month first)
    "%d-%b-%Y",    # DD-Mon-YYYY
    "%Y/%m/%d",    # YYYY/MM/DD
    "%B %d, %Y",   # Month D, YYYY
)

PLAN_MAP = {...}
CONTRACT_TYPE_MAP = {...}
CHURNED_MAP = {...}

OUTPUT_COLUMNS = [...]


def clean_text(value):
    """R1: strip whitespace; map known missing tokens to None."""
    if value is None:
        return None
    text = str(value).strip()
    if text.lower() in MISSING_TOKENS:
        return None
    return text


def parse_signup_date(text):
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(text, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None


def parse_number(text):
    try:
        value = float(text)
    except (TypeError, ValueError):
        return None
    if not math.isfinite(value):
        return None
    return value


def parse_whole_number(text):
    value = parse_number(text)
    if value is None or not value.is_integer():
        return None
    return int(value)


def parse_monthly_fee(text):
    cleaned = re.sub(r"[$,]", "", text)
    cleaned = re.sub(r"usd", "", cleaned, flags=re.IGNORECASE)
    cleaned = cleaned.strip()
    if not cleaned:
        return None
    return parse_number(cleaned)


def parse_tenure_months(text):
    cleaned = re.sub(r"\s*months?\s*$", "", text, flags=re.IGNORECASE).strip()
    if not cleaned:
        return None
    return parse_whole_number(cleaned)
