import os
import re
import json
import math
from datetime import datetime

import pandas as pd


MISSING_TOKENS = {"", "na", "n/a", "null", "none", "-"}

DATE_FORMATS = (
    "%Y-%m-%d",   # YYYY-MM-DD
    "%m/%d/%Y",   # MM/DD/YYYY (month first)
    "%d-%b-%Y",   # DD-Mon-YYYY
    "%Y/%m/%d",   # YYYY/MM/DD
    "%B %d, %Y",  # Month D, YYYY
)

PLAN_MAP = {
    "basic": "Basic",
    "basic plan": "Basic",
    "standard": "Standard",
    "std": "Standard",
    "standard plan": "Standard",
    "premium": "Premium",
    "prem": "Premium",
    "premium plan": "Premium",
}

CONTRACT_MAP = {
    "monthly": "Monthly",
    "month-to-month": "Monthly",
    "m2m": "Monthly",
    "annual": "Annual",
    "yearly": "Annual",
    "1 year": "Annual",
}

CHURN_MAP = {
    "yes": 1, "y": 1, "true": 1, "1": 1, "churned": 1,
    "no": 0, "n": 0, "false": 0, "0": 0, "active": 0,
}

OUTPUT_COLUMNS = [
    "customer_id", "signup_date", "plan", "contract_type", "monthly_fee",
    "tenure_months", "support_tickets", "satisfaction_score", "churned",
    "lifetime_value",
]


def normalize(value):
    """R1: strip whitespace; map NA-like tokens to None."""
    if value is None:
        return None
    text = str(value).strip()
    if text.lower() in MISSING_TOKENS:
        return None
    return text


def parse_signup_date(text):
    for fmt in DATE_FORMATS:
        try:
            return datetime.strptime(text, fmt).strftime("%Y-%m-%d")
        except ValueError:
            continue
    return None


def parse_number(text):
    try:
        value = float(text)
    except (TypeError, ValueError):
        return None
    if not math.isfinite(value):
        return None
    return value


def parse_monthly_fee(text):
    cleaned = re.sub(r"usd", "", text, flags=re.IGNORECASE)
    cleaned = cleaned.replace("$", "").replace(",", "").strip()
    value = parse_number(cleaned)
    if value is None or value <= 0 or value > 500:
        return None
    return value


def parse_tenure_months(text):
    cleaned = re.sub(r"\s*months?\s*$", "", text, flags=re.IGNORECASE).strip()
    value = parse_number(cleaned)
    if value is None or not value.is_integer() or value < 0 or value > 120:
        return None
    return int(value)


def parse_support_tickets(text):
    value = parse_number(text)
    if value is None or not value.is_integer() or value < 0 or value > 100:
        return None
    return int(value)


def parse_satisfaction(text):
    value = parse_number(text)
    if value is None or not value.is_integer() or value < 1 or value > 5:
        return None
    return int(value)


def main():
    input_dir = os.environ["INPUT_DIR"]
    output_dir = os.environ["OUTPUT_DIR"]
    raw_path = os.path.join(input_dir, "raw.csv")

    raw = pd.read_csv(raw_path, dtype=str, keep_default_na=False, na_filter=False)
    raw.columns = [str(c).strip() for c in raw.columns]
    raw_rows = len(raw)

    # R1: normalize every cell
    for col in raw.columns:
        raw[col] = raw[col].map(normalize)

    summary = {"raw_rows": raw_rows}

    # R2: drop duplicate customer_id, keep first occurrence
    missing_id = int(raw["customer_id"].isna().sum())
    raw = raw[raw["customer_id"].notna()]
    before = len(raw)
    raw = raw.drop_duplicates(subset="customer_id", keep="first")
    summary["duplicates_removed"] = before - len(raw)
    summary["missing_customer_id_removed"] = missing_id

    records = []
    counts = {
        "dropped_signup_date": 0,
        "dropped_plan": 0,
        "dropped_contract_type": 0,
        "dropped_monthly_fee": 0,
        "dropped_tenure_months": 0,
        "dropped_support_tickets": 0,
        "dropped_churned": 0,
        "satisfaction_set_missing": 0,
    }

    for row in raw.itertuples(index=False):
        ...
