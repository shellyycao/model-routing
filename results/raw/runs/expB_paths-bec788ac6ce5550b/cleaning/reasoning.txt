def clean_price(val):
    if pd.isna(val):
        return np.nan
    s = str(val)

def parse_discount(val):
    if pd.isna(val):
        return 0.0
    s = str(val).strip()
    if s.endswith('%'):
        s = s[:-1]
    s = s.strip()
    try:
        f = float(s)
        return f
    except:
        return np.nan
df['discount_pct_num'] = df['discount_pct'].apply(parse_discount)
